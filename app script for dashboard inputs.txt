const sheetId = '1Eob3yiujIi0sNtsPQr_HCvQtlybXnq6K11bm1wAcqAo'; // Your Spreadsheet ID
const sheetName = 'input'; // Your sheet name

function doPost(e) {
  const lock = LockService.getScriptLock();
  lock.tryLock(10000); // Locks script for 10 seconds to prevent concurrent access issues

  try {
    const doc = SpreadsheetApp.openById(sheetId);
    const sheet = doc.getSheetByName(sheetName);

    // Define expected headers
    const headers = [
      'LP ID', 'LP Name', 'LP Mobile', 'ALP ID', 'ALP Name', 'ALP Mobile', 
      'Train No', 'LOCO No', 'Sign On Date', 'Sign On Time', 'Direction', 
      'Last Passed Station', 'ML/CIC', 'Submitted By', 'TimeStamp'
    ];

    // Set headers if the first row is blank
    if (sheet.getRange(1, 1).isBlank()) {
      sheet.getRange(1, 1, 1, headers.length).setValues([headers]);
    }

    // Get current headers to map dynamically
    const currentHeaders = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0];
    const nextRow = sheet.getLastRow() + 1;

    const newRow = currentHeaders.map(function(header) {
      switch(header) {
        case 'LP ID':
          return e.parameter['lp-id'] || '';
        case 'LP Name':
          return e.parameter['lp-name-hidden'] || '';
        case 'LP Mobile':
          return e.parameter['lp-mobile-hidden'] || '';
        case 'ALP ID':
          return e.parameter['alp-id'] || '';
        case 'ALP Name':
          return e.parameter['alp-name-hidden'] || '';
        case 'ALP Mobile':
          return e.parameter['alp-mobile-hidden'] || '';
        case 'Train No':
          return e.parameter['train-no'] || '';
        case 'LOCO No':
          return e.parameter['loco-no'] || '';
        case 'Sign On Date':
          return e.parameter['sign-on-date'] || '';
        case 'Sign On Time':
          return e.parameter['sign-on-time'] || '';
        case 'Direction':
          return e.parameter['direction'] || '';
        case 'Last Passed Station':
          return e.parameter['last-passed-station'] || '';
        case 'ML/CIC':
          return e.parameter['ml-cic'] || '';
        case 'Submitted By':
          return e.parameter['user-id'] || '';
        case 'TimeStamp':
          return e.parameter['timestamp'] || formatTimestamp(new Date()); // Use client-side timestamp if provided
        default:
          return ''; // Default empty for unhandled headers
      }
    });

    sheet.getRange(nextRow, 1, 1, newRow.length).setValues([newRow]);

    return ContentService
      .createTextOutput(JSON.stringify({ 'result': 'success', 'row': nextRow }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (e) {
    return ContentService
      .createTextOutput(JSON.stringify({ 'result': 'error', 'error': e.message }))
      .setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock(); // Releases the lock after execution
  }
}

function formatTimestamp(date) {
  const day = String(date.getDate()).padStart(2, '0');
  const month = String(date.getMonth() + 1).padStart(2, '0');
  const year = date.getFullYear();
  const hours = String(date.getHours()).padStart(2, '0');
  const minutes = String(date.getMinutes()).padStart(2, '0');
  const seconds = String(date.getSeconds()).padStart(2, '0');
  return `${day}/${month}/${year} ${hours}:${minutes}:${seconds}`;
}
